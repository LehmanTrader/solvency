<script>
  import {
    quoteRig, subscriptionBreakEven, breakEvenScan,
    type RigSpecV1, type RigComponentV1, type RigQuoteV1,
    type PowerDrawMethod, type ThroughputMethod, type LocalPassRateBasis,
    type SubscriptionScaling,
  } from '../lib/rig-cost.ts';
  import type { BuildInputOrigin } from '../lib/build-cost.ts';
  import { RIG_CATALOG, ELECTRICITY_DEFAULT, catalogEntryToRigSpec, referenceRigById } from '../lib/rig-catalog.ts';
  import { observeClerkAuth, authenticatedJsonFetch, type ClerkAuthState } from '../lib/clerk-client.ts';

  const byId = <T extends HTMLElement>(id: string) => document.getElementById(id) as T;
  const setText = (id: string, next: string) => { const target = byId(id); if (target.textContent !== next) target.textContent = next; };
  const money = (n: number | null) => n === null ? 'Missing' : `$${n.toLocaleString(undefined, { minimumFractionDigits: n < 1 ? 3 : 2, maximumFractionDigits: n < 1 ? 4 : 2 })}`;
  const value = (id: string) => (byId<HTMLInputElement>(id).value ?? '').normalize('NFC').trim();
  const numericInput = (input: HTMLInputElement) => input.value === '' ? Number.NaN : input.valueAsNumber;
  const number = (id: string) => numericInput(byId<HTMLInputElement>(id));

  const form = byId<HTMLFormElement>('rig-form');
  const clerkEnabled = form.dataset.clerkEnabled === 'true';
  const CHECKOUT_UI_ENABLED = form.dataset.checkoutUiEnabled === 'true';

  type FieldMeta = { assertionOrigin?: BuildInputOrigin; sourceUrl?: string; lastVerified?: string };
  interface ComponentRow extends FieldMeta { componentId: string; label: string; condition: 'new' | 'used'; priceUsd: number; }

  let selectedRigId = RIG_CATALOG[0].id;
  let componentRows: ComponentRow[] = [];
  const resaleMeta: FieldMeta = {};
  const powerMeta: FieldMeta = {};
  const electricityMeta: FieldMeta = { ...ELECTRICITY_DEFAULT };
  const throughputMeta: FieldMeta = {};
  let componentSeq = 0;

  // ---- Launch-day demo-mode soft gating (mirrors build-planner.astro's pattern in this page's
  // own script scope: Astro bundles each page's <script> separately, so this defines its own
  // demoTier/demoLocked/DEMO_GATED_SELECTOR/DEMO_GATE_DESCRIPTIONS rather than importing
  // build-planner's). Free: the four reference-rig buttons. Pro: every editable control below. ----
  type DemoTier = 'unknown' | 'free' | 'pro';
  let demoTier: DemoTier = 'unknown';
  let demoEntitlementRequest: AbortController | null = null;
  let demoAuthGeneration = 0;
  const demoLocked = () => CHECKOUT_UI_ENABLED && demoTier !== 'pro';
  const demoGateEl = CHECKOUT_UI_ENABLED ? byId<HTMLElement>('rig-demo-gate') : null;

  function isEntitlementPayload(input: unknown): input is { data: { tier: 'free' | 'pro'; active: boolean } } {
    if (!input || typeof input !== 'object') return false;
    const data = (input as Record<string, unknown>).data;
    if (!data || typeof data !== 'object') return false;
    const d = data as Record<string, unknown>;
    return (d.tier === 'free' || d.tier === 'pro') && typeof d.active === 'boolean';
  }

  function renderDemoGateState() {
    if (!CHECKOUT_UI_ENABLED) return;
    byId<HTMLElement>('local-hardware').classList.toggle('demo-gate-locked', demoLocked());
    const pro = demoTier === 'pro';
    setText('rig-access-badge', pro ? 'PRO · full rig editor' : 'FREE DEMO · reference rigs');
    setText('rig-access-title', pro
      ? 'Pro is active. Every rig control and the cloud break-even below are unlocked.'
      : 'Free demo — view the four reference rigs at their catalog defaults. Pro unlocks every control, a supplied pass rate and the break-even comparison.');
    byId<HTMLElement>('rig-cloud-locked').hidden = !demoLocked();
    byId<HTMLElement>('rig-cloud-unlocked').hidden = demoLocked();
    render();
  }

  async function refreshDemoEntitlement() {
    if (!CHECKOUT_UI_ENABLED) return;
    demoAuthGeneration += 1;
    const generation = demoAuthGeneration;
    demoEntitlementRequest?.abort();
    const controller = new AbortController();
    demoEntitlementRequest = controller;
    try {
      const payload = await authenticatedJsonFetch('/api/entitlement', { signal: controller.signal });
      if (generation !== demoAuthGeneration) return;
      demoTier = isEntitlementPayload(payload) && payload.data.tier === 'pro' && payload.data.active ? 'pro' : 'free';
    } catch {
      // Any transport failure, 503, signed-out or non-Pro response fails closed to Free — never unlocked.
      if (generation !== demoAuthGeneration) return;
      demoTier = 'free';
    } finally {
      if (generation === demoAuthGeneration) { demoEntitlementRequest = null; renderDemoGateState(); }
    }
  }

  function handleDemoAuth(state: ClerkAuthState) {
    if (!CHECKOUT_UI_ENABLED || state.status === 'checking') return;
    void refreshDemoEntitlement();
  }

  function showDemoGatePrompt(detail: string) {
    if (!CHECKOUT_UI_ENABLED || !demoGateEl) return;
    byId('rig-demo-gate-detail').textContent = detail;
    demoGateEl.hidden = false;
  }
  function hideDemoGatePrompt() { if (demoGateEl) demoGateEl.hidden = true; }

  const DEMO_GATE_DESCRIPTIONS: Record<string, string> = {
    'rig-add-component': 'Add a component to this rig.',
    'passrate-enabled': 'Supply a pass rate for cost per solved task.',
    'cloud-cap-enabled': 'Model a capped subscription tier.',
  };
  function demoGateDetail(target: Element): string {
    const field = (target as HTMLElement).dataset?.field;
    if (field && DEMO_GATE_DESCRIPTIONS[field]) return DEMO_GATE_DESCRIPTIONS[field];
    if (target.id && DEMO_GATE_DESCRIPTIONS[target.id]) return DEMO_GATE_DESCRIPTIONS[target.id];
    return 'Edit this rig assumption.';
  }
  const DEMO_GATED_SELECTOR = ['#rig-add-component', '[data-action="remove-component"]'].join(', ');

  function handleDemoGateClick(event: MouseEvent) {
    if (!demoLocked()) return;
    const target = (event.target as Element).closest<HTMLElement>(DEMO_GATED_SELECTOR);
    if (!target) return;
    if (target.id === 'rig-add-component' && componentRows.length === 0) return; // never block down to zero components
    event.preventDefault();
    event.stopPropagation();
    showDemoGatePrompt(demoGateDetail(target));
  }

  /** Every scalar `[data-field]` input in the rig form reverts on edit while locked, and shows the
   * upsell toast — the same technique build-planner.astro uses for its sensitivity controls,
   * generalized across every rig field since the whole editor is a Pro control here. */
  function guardAllRigFields() {
    form.querySelectorAll<HTMLInputElement | HTMLSelectElement>('[data-field]').forEach((el) => {
      let lastGoodValue = el.value;
      const guard = (event: Event) => {
        if (!demoLocked()) { lastGoodValue = el.value; return; }
        event.stopPropagation();
        if (el.value !== lastGoodValue) el.value = lastGoodValue;
        showDemoGatePrompt(demoGateDetail(el));
      };
      el.addEventListener('input', guard);
      el.addEventListener('change', guard);
    });
  }

  // ---- Catalog selection ----
  function resetComponentRowsFromCatalog(components: RigComponentV1[]) {
    componentRows = components.map((c) => ({
      componentId: c.componentId, label: c.label, condition: c.condition, priceUsd: c.priceUsd,
      assertionOrigin: c.assertionOrigin, sourceUrl: c.sourceUrl, lastVerified: c.lastVerified,
    }));
  }

  function selectRig(id: string) {
    const entry = referenceRigById(id);
    if (!entry) return;
    selectedRigId = id;
    byId<HTMLInputElement>('rig-name').value = entry.name;
    resetComponentRowsFromCatalog(entry.components);
    byId<HTMLInputElement>('rig-resale-usd').value = String(entry.resale.residualUsd);
    Object.assign(resaleMeta, { assertionOrigin: entry.resale.assertionOrigin, sourceUrl: entry.resale.sourceUrl, lastVerified: entry.resale.lastVerified });
    byId<HTMLInputElement>('rig-horizon-months').value = String(entry.defaultHorizonMonths);
    byId<HTMLInputElement>('rig-power-watts').value = String(entry.power.systemWatts);
    byId<HTMLSelectElement>('rig-power-method').value = entry.power.method;
    Object.assign(powerMeta, { assertionOrigin: entry.power.assertionOrigin, sourceUrl: entry.power.sourceUrl, lastVerified: entry.power.lastVerified });
    byId<HTMLInputElement>('rig-throughput-tps').value = String(entry.throughput.tokensPerSecond);
    byId<HTMLInputElement>('rig-throughput-runtime').value = entry.throughput.runtime;
    byId<HTMLInputElement>('rig-throughput-runtime-version').value = entry.throughput.runtimeVersion ?? '';
    byId<HTMLInputElement>('rig-throughput-context').value = entry.throughput.contextTokens === null ? '' : String(entry.throughput.contextTokens);
    byId<HTMLSelectElement>('rig-throughput-method').value = entry.throughput.method;
    Object.assign(throughputMeta, { assertionOrigin: entry.throughput.assertionOrigin, sourceUrl: entry.throughput.sourceUrl, lastVerified: entry.throughput.lastVerified });
    byId<HTMLInputElement>('rig-model-name').value = entry.model.modelName;
    byId<HTMLInputElement>('rig-model-quant').value = entry.model.quantization ?? '';
    byId<HTMLInputElement>('rig-passrate-enabled').checked = false;
    byId<HTMLElement>('rig-passrate-fields').hidden = true;

    document.querySelectorAll<HTMLButtonElement>('[data-rig-option]').forEach((btn) => {
      const active = btn.dataset.rigId === id;
      btn.setAttribute('aria-pressed', active ? 'true' : 'false');
      btn.classList.toggle('card-active', active);
    });
    const primary = entry.components[0];
    const src = byId<HTMLElement>('rig-picker-source');
    src.replaceChildren();
    if (primary.sourceUrl) {
      const a = document.createElement('a'); a.className = 'link small-sans'; a.href = primary.sourceUrl; a.textContent = primary.sourceUrl;
      src.append('Primary component price: ', a, ` · verified ${primary.lastVerified}`);
    }
    renderComponentRows();
    renderFieldSources();
    render();
  }

  function componentRowEl(row: ComponentRow): HTMLElement {
    const wrap = document.createElement('div');
    wrap.className = 'grid gap-4 sm:grid-cols-[1fr_10rem_8rem_auto] items-end';
    wrap.dataset.componentId = row.componentId;

    const labelWrap = document.createElement('label'); labelWrap.className = 'block';
    labelWrap.innerHTML = '<span class="label">Component</span>';
    const labelInput = document.createElement('input');
    labelInput.className = 'ctl ctl-row mt-2 w-full'; labelInput.dataset.field = 'component-label'; labelInput.value = row.label;
    labelWrap.appendChild(labelInput);

    const priceWrap = document.createElement('label'); priceWrap.className = 'block';
    priceWrap.innerHTML = '<span class="label">Price</span>';
    const priceSpan = document.createElement('span'); priceSpan.className = 'money-input mt-2';
    const dollar = document.createElement('span'); dollar.textContent = '$';
    const priceInput = document.createElement('input');
    priceInput.className = 'ctl ctl-row w-full'; priceInput.type = 'number'; priceInput.min = '0'; priceInput.step = 'any';
    priceInput.dataset.field = 'component-price'; priceInput.value = String(row.priceUsd);
    priceSpan.append(dollar, priceInput); priceWrap.appendChild(priceSpan);

    const conditionWrap = document.createElement('label'); conditionWrap.className = 'block';
    conditionWrap.innerHTML = '<span class="label">Condition</span>';
    const conditionSelect = document.createElement('select');
    conditionSelect.className = 'ctl ctl-row mt-2 w-full'; conditionSelect.dataset.field = 'component-condition';
    conditionSelect.innerHTML = '<option value="new">New</option><option value="used">Used</option>';
    conditionSelect.value = row.condition;
    conditionWrap.appendChild(conditionSelect);

    const removeBtn = document.createElement('button');
    removeBtn.type = 'button'; removeBtn.className = 'btn btn-quiet'; removeBtn.textContent = 'Remove';
    removeBtn.dataset.action = 'remove-component';
    removeBtn.disabled = componentRows.length <= 1;

    const downgrade = () => {
      row.label = labelInput.value; row.priceUsd = priceInput.valueAsNumber; row.condition = conditionSelect.value as 'new' | 'used';
      row.assertionOrigin = 'user_asserted'; row.sourceUrl = undefined; row.lastVerified = undefined;
      render();
    };
    labelInput.addEventListener('input', downgrade);
    priceInput.addEventListener('input', downgrade);
    conditionSelect.addEventListener('change', downgrade);
    removeBtn.addEventListener('click', () => {
      if (demoLocked()) { showDemoGatePrompt('Remove a component from this rig.'); return; }
      if (componentRows.length <= 1) return;
      componentRows = componentRows.filter((c) => c.componentId !== row.componentId);
      renderComponentRows(); render();
    });

    wrap.append(labelWrap, priceWrap, conditionWrap, removeBtn);
    return wrap;
  }

  function renderComponentRows() {
    const container = byId('rig-components');
    container.replaceChildren(...componentRows.map(componentRowEl));
  }

  function addComponent() {
    if (demoLocked()) { showDemoGatePrompt(DEMO_GATE_DESCRIPTIONS['rig-add-component']); return; }
    componentSeq += 1;
    componentRows = [...componentRows, {
      componentId: `custom-${componentSeq}`, label: 'New component', condition: 'new', priceUsd: 0, assertionOrigin: 'user_asserted',
    }];
    renderComponentRows(); render();
  }

  function renderFieldSources() {
    const sourceLine = (meta: FieldMeta) => meta.sourceUrl ? `sourced · verified ${meta.lastVerified}` : 'modelled default — edit freely';
    setText('rig-power-source', sourceLine(powerMeta));
    setText('rig-electricity-source', `${electricityMeta.region ?? ''} ${sourceLine(electricityMeta)}`.trim());
    setText('rig-throughput-source', sourceLine(throughputMeta));
  }

  // ---- Assembling and quoting the current rig ----
  function currentRigSpec(): RigSpecV1 {
    const passRateEnabled = byId<HTMLInputElement>('rig-passrate-enabled').checked;
    const contextRaw = value('rig-throughput-context');
    return {
      schemaVersion: 1,
      name: value('rig-name'),
      components: componentRows.map((c) => ({
        componentId: c.componentId, label: c.label, condition: c.condition, priceUsd: c.priceUsd,
        assertionOrigin: c.assertionOrigin, sourceUrl: c.sourceUrl, lastVerified: c.lastVerified,
      })),
      resale: { residualUsd: number('rig-resale-usd'), ...resaleMeta },
      horizonMonths: number('rig-horizon-months'),
      power: { systemWatts: number('rig-power-watts'), method: byId<HTMLSelectElement>('rig-power-method').value as PowerDrawMethod, ...powerMeta },
      electricity: { usdPerKwh: number('rig-electricity-rate'), region: value('rig-electricity-region') || undefined, ...electricityMeta },
      throughput: {
        tokensPerSecond: number('rig-throughput-tps'),
        runtime: value('rig-throughput-runtime'),
        runtimeVersion: value('rig-throughput-runtime-version') || null,
        contextTokens: contextRaw === '' ? null : Number(contextRaw),
        method: byId<HTMLSelectElement>('rig-throughput-method').value as ThroughputMethod,
        ...throughputMeta,
      },
      model: {
        modelName: value('rig-model-name'),
        quantization: value('rig-model-quant') || null,
        passRate: passRateEnabled ? {
          rate: number('rig-passrate-value') / 100,
          benchmark: value('rig-passrate-benchmark'),
          basis: byId<HTMLSelectElement>('rig-passrate-basis').value as LocalPassRateBasis,
        } : undefined,
      },
      workload: {
        tasksPerMonth: number('rig-workload-tasks'),
        tokensPerAttempt: number('rig-workload-tokens'),
        overheadSecondsPerAttempt: number('rig-workload-overhead'),
      },
    };
  }

  function originLabel(origin: BuildInputOrigin | undefined) {
    return (origin ?? 'user_asserted').replaceAll('_', ' ');
  }

  function renderLedger(quote: RigQuoteV1, spec: RigSpecV1) {
    const ledger = byId('rig-ledger'); ledger.replaceChildren();
    const line = (text: string) => { const li = document.createElement('li'); li.textContent = text; ledger.appendChild(li); };
    for (const c of spec.components) {
      line(`${c.label}: ${money(c.priceUsd)} · ${c.condition} · origin ${originLabel(c.assertionOrigin)}${c.sourceUrl ? ` · ${c.sourceUrl} (verified ${c.lastVerified})` : ''}`);
    }
    line(`Resale residual: ${money(spec.resale.residualUsd)} · origin ${originLabel(spec.resale.assertionOrigin)}`);
    line(`Power: ${spec.power.systemWatts}W · ${spec.power.method.replaceAll('_', ' ')} · origin ${originLabel(spec.power.assertionOrigin)}`);
    line(`Electricity: $${spec.electricity.usdPerKwh}/kWh${spec.electricity.region ? ` · ${spec.electricity.region}` : ''} · origin ${originLabel(spec.electricity.assertionOrigin)}`);
    line(`Throughput: ${spec.throughput.tokensPerSecond} tok/s · ${spec.throughput.runtime}${spec.throughput.runtimeVersion ? ` ${spec.throughput.runtimeVersion}` : ' (version unpinned)'} · ${spec.throughput.method.replaceAll('_', ' ')} · origin ${originLabel(spec.throughput.assertionOrigin)}`);
    line(spec.model.passRate
      ? `Pass rate: ${(spec.model.passRate.rate * 100).toFixed(1)}% on ${spec.model.passRate.benchmark} · basis ${spec.model.passRate.basis.replaceAll('_', ' ')} · origin ${originLabel(spec.model.passRate.assertionOrigin)}`
      : 'Pass rate: not supplied — cost per solved task stays Missing.');
    if (quote.warnings.length) for (const w of quote.warnings) line(`Warning: ${w}`);
  }

  let lastQuote: RigQuoteV1 | null = null;

  function renderCrossover(quote: RigQuoteV1) {
    const headline = byId('rig-crossover-headline');
    const detail = byId('rig-crossover-detail');
    if (demoLocked()) return; // the locked overlay covers this section entirely
    if (!quote.valid || quote.passRate === null || quote.amortizationPerMonthUsd === null || quote.electricityPerAttemptUsd === null) {
      headline.textContent = 'Missing — supply a valid rig and a pass rate above to compare.';
      detail.textContent = '';
      return;
    }
    const cloudUsd = number('cloud-usd-month');
    const cloudPassRatePct = number('cloud-passrate');
    const capEnabled = byId<HTMLInputElement>('cloud-cap-enabled').checked;
    const capRaw = capEnabled ? number('cloud-cap-value') : null;
    const scaling = byId<HTMLSelectElement>('cloud-scaling').value as SubscriptionScaling;
    if (!Number.isFinite(cloudUsd) || !Number.isFinite(cloudPassRatePct) || cloudPassRatePct <= 0 || cloudPassRatePct > 100) {
      headline.textContent = 'Missing — enter a valid subscription price and pass rate.';
      detail.textContent = '';
      return;
    }
    const rigSide = { amortizationPerMonthUsd: quote.amortizationPerMonthUsd, electricityPerAttemptUsd: quote.electricityPerAttemptUsd, passRate: quote.passRate.rate };
    const subscription = { usdPerMonth: cloudUsd, passRate: cloudPassRatePct / 100, tasksPerMonthCap: capEnabled ? capRaw : null, scaling };

    if (!capEnabled) {
      const out = subscriptionBreakEven(rigSide, subscription);
      if (out.kind === 'crossover') {
        headline.textContent = `Above ~${Math.ceil(out.tasksPerMonth).toLocaleString()} tasks/month, this rig beats that subscription per solved task.`;
        detail.textContent = 'Uncapped comparison via subscriptionBreakEven — the closed-form crossover.';
      } else if (out.kind === 'local_always_cheaper') {
        headline.textContent = 'This rig is cheaper than that subscription at every volume in this scenario.';
        detail.textContent = 'Uncapped comparison via subscriptionBreakEven.';
      } else if (out.kind === 'subscription_always_cheaper') {
        headline.textContent = 'That subscription is cheaper at every volume in this scenario.';
        detail.textContent = 'An uncapped flat fee dilutes toward zero with volume; this rig has a nonzero electricity floor per solved task, so it never catches up.';
      } else if (out.kind === 'equal') {
        headline.textContent = 'Tied at every volume in this scenario.';
        detail.textContent = '';
      } else {
        headline.textContent = 'Missing — the inputs on both sides must be valid, finite numbers.';
        detail.textContent = '';
      }
      return;
    }

    const out = breakEvenScan(rigSide, subscription);
    if (out.kind === 'crossover') {
      headline.textContent = `Above ~${Math.ceil(out.tasksPerMonth).toLocaleString()} tasks/month, this rig beats that capped subscription per solved task.`;
      detail.textContent = 'Capped comparison via breakEvenScan (stacked seats above the cap).';
    } else if (out.kind === 'mixed') {
      headline.textContent = `First cheaper around ${Math.ceil(out.firstLocalWin).toLocaleString()} tasks/month, but the seat cap makes this flip more than once.`;
      detail.textContent = 'Capped comparison via breakEvenScan: the subscription’s per-seat cap creates a sawtooth cost curve, so this is not a single stable crossover.';
    } else if (out.kind === 'local_always_cheaper') {
      headline.textContent = 'This rig is cheaper than that capped subscription at every scanned volume.';
      detail.textContent = 'Capped comparison via breakEvenScan.';
    } else if (out.kind === 'subscription_always_cheaper') {
      headline.textContent = 'That capped subscription is cheaper at every scanned volume.';
      detail.textContent = 'Capped comparison via breakEvenScan.';
    } else {
      headline.textContent = 'Missing — the inputs on both sides must be valid, finite numbers.';
      detail.textContent = '';
    }
  }

  function render() {
    if (!componentRows.length) return;
    const spec = currentRigSpec();
    const quote = quoteRig(spec, new Date().toISOString());
    lastQuote = quote;

    setText('rig-build-cost', money(quote.buildCostUsd));
    setText('rig-strip-attempt', money(quote.costPerAttemptUsd));
    const solvedMissing = quote.costPerSolvedTaskUsd === null;
    setText('rig-strip-solved', money(quote.costPerSolvedTaskUsd));
    byId('rig-strip-solved').classList.toggle('missing', solvedMissing);
    setText('rig-strip-solved-note', solvedMissing
      ? (quote.missing.includes('model.passRate') ? 'Supply a pass rate above' : quote.missing.includes('throughput.runtimeVersion') ? 'Supply a runtime version above' : quote.missing.includes('model.quantization') ? 'Supply a quantization above' : 'Missing input above')
      : 'Amortization + electricity ÷ pass rate');
    setText('rig-strip-month', money(quote.monthlyCostUsd));

    const messages = byId('rig-messages'); messages.replaceChildren();
    if (quote.errors.length) {
      const ul = document.createElement('ul'); ul.className = 'small list-disc pl-5 text-[var(--color-worse)]';
      for (const e of quote.errors) { const li = document.createElement('li'); li.textContent = e; ul.appendChild(li); }
      messages.appendChild(ul);
    } else {
      const p = document.createElement('p'); p.className = 'small';
      p.textContent = quote.warnings.length ? 'Quote valid, with warnings below.' : 'Quote valid.';
      messages.appendChild(p);
    }

    renderLedger(quote, spec);
    renderCrossover(quote);
  }

  // ---- Wiring ----
  document.getElementById('rig-picker')!.addEventListener('click', (event) => {
    const btn = (event.target as Element).closest<HTMLButtonElement>('[data-rig-option]');
    if (!btn?.dataset.rigId) return;
    selectRig(btn.dataset.rigId);
  });
  byId('rig-add-component').addEventListener('click', addComponent);
  byId<HTMLInputElement>('rig-passrate-enabled').addEventListener('change', (event) => {
    if (demoLocked()) { (event.target as HTMLInputElement).checked = false; showDemoGatePrompt(DEMO_GATE_DESCRIPTIONS['passrate-enabled']); return; }
    byId<HTMLElement>('rig-passrate-fields').hidden = !(event.target as HTMLInputElement).checked;
    render();
  });
  byId<HTMLInputElement>('cloud-cap-enabled').addEventListener('change', (event) => {
    if (demoLocked()) { (event.target as HTMLInputElement).checked = false; showDemoGatePrompt(DEMO_GATE_DESCRIPTIONS['cloud-cap-enabled']); return; }
    byId<HTMLElement>('cloud-cap-fields').hidden = !(event.target as HTMLInputElement).checked;
    render();
  });
  form.addEventListener('input', (event) => {
    const field = (event.target as HTMLElement).dataset?.field;
    if (field && field.startsWith('cloud-')) render();
    else if (field && !field.startsWith('component-') && field !== 'passrate-enabled' && field !== 'cloud-cap-enabled') render();
  });
  form.addEventListener('change', (event) => {
    const field = (event.target as HTMLElement).dataset?.field;
    if (field === 'power-method' || field === 'throughput-method' || field === 'passrate-basis' || field === 'cloud-scaling') render();
  });

  if (CHECKOUT_UI_ENABLED) {
    form.addEventListener('click', handleDemoGateClick, true);
    guardAllRigFields();
    byId<HTMLButtonElement>('rig-demo-gate-dismiss').addEventListener('click', hideDemoGatePrompt);
    document.addEventListener('keydown', (event) => { if (event.key === 'Escape') hideDemoGatePrompt(); });
    const stopDemoAuth = observeClerkAuth(handleDemoAuth, { enabled: clerkEnabled });
    addEventListener('pagehide', () => { stopDemoAuth(); demoEntitlementRequest?.abort(); }, { once: true });
    renderDemoGateState();
  } else {
    byId<HTMLElement>('rig-cloud-locked').hidden = true;
    byId<HTMLElement>('rig-cloud-unlocked').hidden = false;
  }

  selectRig(selectedRigId);
</script>
